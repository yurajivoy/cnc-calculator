    <script>

        const materialsData = {

            plywood: { name: "Фанера", rpm: 18000, fzBase: 0.09, depthFactor: 0.8 },

            mdf:     { name: "МДФ",     rpm: 18000, fzBase: 0.10, depthFactor: 1.0 },

            pine:    { name: "Сосна",   rpm: 20000, fzBase: 0.12, depthFactor: 1.2 },

            walnut:  { name: "Орех",    rpm: 17000, fzBase: 0.08, depthFactor: 0.7 },

            oak:     { name: "Дуб",     rpm: 16000, fzBase: 0.06, depthFactor: 0.5 }

        };


        const diamInput = document.getElementById('diam');

        const thickInput = document.getElementById('thick');

        const materialSelect = document.getElementById('material');


        const valDiam = document.getElementById('val-diam');

        const valThick = document.getElementById('val-thick');


        const resFeed = document.getElementById('res-feed');

        const resRpm = document.getElementById('res-rpm');

        const resPasses = document.getElementById('res-passes');

        const resDepthPass = document.getElementById('res-depth-pass');

        const resFz = document.getElementById('res-fz');


        function calculate() {

            const d = parseFloat(diamInput.value);

            const t = parseFloat(thickInput.value);

            const matKey = materialSelect.value;

            const mat = materialsData[matKey];


            valDiam.innerText = d + ' мм';

            valThick.innerText = t + ' мм';


            // Коррекция подачи на зуб в зависимости от диаметра фрезы (тонкие фрезы требуют меньшей нагрузки)

            let fz = mat.fzBase * (d / 8); 

            fz = Math.max(0.03, Math.min(0.15, fz));


            // Обороты шпинделя (3 зуба, z = 3)

            const rpm = mat.rpm;

            const feed = Math.round(rpm * 3 * fz);


            // Расчет глубины за проход

            let maxAp = d * mat.depthFactor;

            maxAp = Math.max(2, Math.min(maxAp, t));

            

            const passes = Math.ceil(t / maxAp);

            const actualAp = (t / passes).toFixed(2);


            // Вывод результатов

            resFeed.innerHTML = `${feed.toLocaleString()} <span class="text-sm font-normal text-slate-400">мм/мин</span>`;

            resRpm.innerHTML = `${rpm.toLocaleString()} <span class="text-sm font-normal text-slate-400">об/мин</span>`;

            resPasses.innerText = `${passes} ${passes === 1 ? 'проход' : passes < 5 ? 'прохода' : 'проходов'}`;

            resDepthPass.innerText = `${actualAp} мм / проход (макс. ${maxAp.toFixed(1)} мм)`;

            resFz.innerHTML = `${fz.toFixed(3)} <span class="text-sm font-normal text-slate-400">мм/зуб</span>`;

        }


        diamInput.addEventListener('input', calculate);

        thickInput.addEventListener('input', calculate);

        materialSelect.addEventListener('change', calculate);


        calculate();

    </script>
